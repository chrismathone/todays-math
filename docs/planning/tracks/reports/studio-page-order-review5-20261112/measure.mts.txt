async function open(file: string) {
  const bytes = await fs.readFile(path.join(directory, file));
  const pdf = await pdfjs.getDocument({
    data: new Uint8Array(bytes), useSystemFonts: false,
    standardFontDataUrl: path.join(pdfRoot, "standard_fonts") + path.sep,
    wasmUrl: path.join(pdfRoot, "wasm") + path.sep,
  }).promise;
  return { pdf, bytes };
}

function objects(page: any) {
  // Read-only instrumentation of actual PDFPageProxy.objs, not process RSS.
  // Installed PDFObjects exposes a resolved-object iterator; data is not copied here.
  const images = Array.from(page.objs as Iterable<[string, any]>).flatMap(([id, object]) =>
    object?.width && object?.height && object?.data?.byteLength
      ? [{ id, width: object.width, height: object.height, decodedBytes: object.data.byteLength }]
      : [],
  );
  return { resolvedObjects: Array.from(page.objs).length, images,
    decodedBytes: images.reduce((total, image) => total + image.decodedBytes, 0) };
}

function instrument(page: any, cleanupEvents: any[], renderEvents: any[]) {
  return new Proxy(page, { get(target, key) {
    if (key === "cleanup") return (...args: any[]) => {
      const before = objects(target);
      const success = target.cleanup(...args);
      cleanupEvents.push({ page: page.pageNumber, success, before, after: objects(target) });
      return success;
    };
    if (key === "render") return (...args: any[]) => {
      const start = performance.now();
      const task = target.render(...args);
      task.promise.then(() => renderEvents.push({ page: page.pageNumber,
        scale: args[0].viewport.scale, elapsedMs: performance.now() - start,
        retained: objects(target) }), () => {});
      return task;
    };
    const value = Reflect.get(target, key);
    return typeof value === "function" ? value.bind(target) : value;
  } });
}

const { pdf, bytes } = await open("white-scan-16.pdf");
const pages: PDFPageProxy[] = [];
const rows = [];
const cleanupEvents: any[] = [];
const renderEvents: any[] = [];
const start = performance.now();
for (let number = 1; number <= pdf.numPages; number++) {
  const page = await pdf.getPage(number);
  pages.push(page);
  const wrapped = instrument(page, cleanupEvents, renderEvents);
  const wrappedPdf = { getPage: async () => wrapped };
  const published: any[] = [];
  const preview = await renderStudioPagePreview({ pdf: wrappedPdf as any }, number,
    new AbortController().signal, { onPreview: (preview: any) => {
      published.push({ urlReady: preview.url.startsWith("data:image/"),
        inkRatio: preview.inkRatio, autoOut: preview.autoOut ?? null,
        elapsedMs: performance.now() - start });
    } });
  rows.push({ page: number, inkRatio: preview.inkRatio, autoOut: preview.autoOut ?? null,
    error: preview.error ?? null, retainedAfterInvocation: objects(page), published });
}
const total = () => pages.reduce((sum, page) => sum + objects(page).decodedBytes, 0);
const retainedDecodedBytesAfterPreviews = total();
const elapsedMs = performance.now() - start;
await fs.writeFile(path.join(directory, `${mode}-scan.json`), JSON.stringify({
  mode, sourceSha256: hash(source), fixtureBytes: bytes.length,
  retainedDecodedBytesAfterPreviews, elapsedMs, rows, cleanupEvents, renderEvents,
}, null, 2) + "\n");
const explicitCleanup = pages.map((page) => ({ page: page.pageNumber,
  success: page.cleanup(), after: objects(page) }));

async function region(page: any) {
  const canvas = createCanvas(230, 190);
  const viewport = page.getViewport({ scale: 1.7 });
  await page.render({ canvas, canvasContext: canvas.getContext("2d"), viewport,
    transform: [1, 0, 0, 1, -90, -140] }).promise;
  const data = canvas.getContext("2d").getImageData(0, 0, canvas.width, canvas.height).data;
  let nonwhitePixels = 0;
  for (let i = 0; i < data.length; i += 4)
    if (data[i] < 255 || data[i + 1] < 255 || data[i + 2] < 255) nonwhitePixels++;
  const result = { sha256: hash(data), width: canvas.width, height: canvas.height, nonwhitePixels };
  canvas.width = canvas.height = 0;
  return result;
}
// Revisit the exact shared scan document after all page caches have been released.
const revisitedPage = await pdf.getPage(1);
const scanRegionBefore = await region(revisitedPage);
assert(revisitedPage.cleanup());
const scanRegionAfter = await region(await pdf.getPage(1));
assert.deepEqual(scanRegionAfter, scanRegionBefore);
assert(revisitedPage.cleanup());
await pdf.loadingTask.destroy();

// Exercise real supported deferred cleanup: pause a renderer at its public onContinue.
const activeCanvas = createCanvas(595, 842);
let resume!: () => void;
let pausedResolve!: () => void;
const paused = new Promise<void>((resolve) => { pausedResolve = resolve; });
const active = probePage.render({ canvas: activeCanvas as unknown as HTMLCanvasElement,
  canvasContext: activeCanvas.getContext("2d") as unknown as CanvasRenderingContext2D,
  viewport: probePage.getViewport({ scale: 1 }) });
active.onContinue = (continueRender: () => void) => {
  active.onContinue = null as unknown as typeof active.onContinue;
  resume = continueRender;
  pausedResolve();
};
await paused;
const deferredCleanupResult = probePage.cleanup();
assert.equal(deferredCleanupResult, false);
resume();
await active.promise;
const afterDeferredCleanup = objects(probePage);
assert.equal(afterDeferredCleanup.decodedBytes, 0);
assert.equal(afterDeferredCleanup.resolvedObjects, 0);
activeCanvas.width = activeCanvas.height = 0;
await probePdf.loadingTask.destroy();
