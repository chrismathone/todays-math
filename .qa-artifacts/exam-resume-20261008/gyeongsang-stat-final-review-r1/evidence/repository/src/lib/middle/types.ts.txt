import type { Difficulty } from "@/contracts/common.contract";

import type { Rng } from "../elementary/types";

/**
 * 중학 유형 엔진이 낸 한 문항. 정답은 생성기가 같은 숫자로 계산한다 (AI 아님).
 *
 * `facts` 는 생성기가 문항을 만들 때 쓴 **원 수치**다 — 테스트가 발문·정답을
 * 재계산으로 검산할 때 쓴다(동어반복 방지: 검산식은 테스트 쪽에 따로 있다).
 * DB 적재 대상이 아니다.
 */
export type MidProblem = {
  grade: string;
  /** 우리 `Unit.section` 정확한 이름 — 적재 시 단원 매칭 키 */
  unitSection: string;
  rpmBook: string;
  rpmMinor: string;
  rpmType: string;
  rpmTitle: string;
  content: string;
  answer: string;
  solution: string;
  difficulty: Difficulty;
  facts: Record<string, number>;
  /** 생성기가 직접 준 오답 넷 — 없으면 공용 객관식 파생기가 만든다. */
  distractors?: readonly string[];
  /**
   * 도형이 필요한 유형만 채운다 — FigureSpec v2 (09-figure-engine-guide §1 계층 A).
   * 실증 완료 목록(§2)에 있는 도형만 쓸 것. 없는 도형이 필요하면 그 유형을
   * 보류 목록에 남기고 지어내지 말 것.
   */
  figureSpec?: Record<string, unknown> | null;
};

export type TypeHandler = (rng: Rng) => MidProblem;

/** 유형 등록 한 건 — 색인 메타 + 생성기 */
export type TypeEntry = {
  /** `책|중단원|번호` — 예: `"1-1|일차방정식의 활용|01"` */
  key: string;
  grade: string;
  unitSection: string;
  rpmBook: string;
  rpmMinor: string;
  rpmType: string;
  rpmTitle: string;
  handler: TypeHandler;
};
