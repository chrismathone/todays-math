/** 조건을 만족할 때까지 다시 뽑는다 — 못 찾으면 던진다(조용히 기본값 금지). */
function attempt<T>(tries: number, f: () => T | null): T {
  for (let i = 0; i < tries; i += 1) {
    const v = f();
    if (v !== null) return v;
  }
  throw new Error("유형 생성 재시도 초과 — 매개변수 범위를 확인하십시오");
}

function gcd(a: number, b: number): number {
  return b === 0 ? a : gcd(b, a % b);
}

function r2(v: number): number {
  return Math.round(v * 100) / 100;
}

/* ────────────────────────────────────────────────────────────────────────
 * 정확한 무리수 대수 — n√r / d
 * ──────────────────────────────────────────────────────────────────────── */

/** `n\sqrt{r}/d` 꼴. `r` 은 제곱인수를 뺀 값, `d>0`, 부호는 `n` 이 갖는다. */
type Surd = { n: number; r: number; d: number };

/** 제곱인수를 밖으로 빼고 약분한다. 정규형이 하나여야 `+`/`-` 를 가를 수 있다. */
function surd(n: number, r: number, d: number): Surd {
  if (d === 0) throw new Error("무리수 분모가 0입니다");
  let nn = n;
  let rr = r;
  let dd = d;
  if (dd < 0) {
    nn = -nn;
    dd = -dd;
  }
  if (rr < 1) throw new Error(`근호 안이 ${rr} 입니다`);
  for (let k = 2; k * k <= rr; k += 1) {
    while (rr % (k * k) === 0) {
      rr /= k * k;
      nn *= k;
    }
  }
  if (nn === 0) return { n: 0, r: 1, d: 1 };
  const g = gcd(Math.abs(nn), dd);
  return { n: nn / g, r: rr, d: dd / g };
}

function surdMul(a: Surd, b: Surd): Surd {
  return surd(a.n * b.n, a.r * b.r, a.d * b.d);
}

/**
 * `a / b` — 분모를 유리화한다.
 * `(n1√r1/d1) ÷ (n2√r2/d2) = n1·d2·√(r1·r2) / (d1·n2·r2)`.
 *
 * ⚠️ 2026-08-23: 처음 판은 `d2`·`r2` 를 빠뜨려 **`d2 === r2` 인 45° 계열에서만
 * 우연히 맞았다.** `20 ÷ cos 30°` 가 `20√3` 으로 나왔고(참값 `40√3/3`),
 * 시험이 아니라 **미리보기를 눈으로 보다** 드러났다. 시험은 이 함수를 안 쓰고
 * 부동소수 삼각함수로 답을 다시 유도하므로 이제는 빨개진다.
 */
function surdDiv(a: Surd, b: Surd): Surd {
  if (b.n === 0) throw new Error("무리수를 0으로 나눌 수 없습니다");
  return surd(a.n * b.d, a.r * b.r, a.d * b.n * b.r);
}

/** 근호가 같을 때만 더한다 — 다르면 `null`(호출자가 그 조합을 버린다). */
function surdAdd(a: Surd, b: Surd): Surd | null {
  if (a.n === 0) return b;
  if (b.n === 0) return a;
  if (a.r !== b.r) return null;
  return surd(a.n * b.d + b.n * a.d, a.r, a.d * b.d);
}

function surdSub(a: Surd, b: Surd): Surd | null {
  return surdAdd(a, { n: -b.n, r: b.r, d: b.d });
}

function surdValue(s: Surd): number {
  return (s.n * Math.sqrt(s.r)) / s.d;
}

/**
 * **순수 숫자 분수는 진분수만** 낸다 — 가분수는 지면에서 대분수로 바뀐다.
 * 어기면 던진다(조용히 `1\frac{1}{2}` 가 나가는 것을 막는다).
 */
function frac(n: number, d: number): string {
  if (d === 0) throw new Error("분모가 0입니다");
  const sign = n * d < 0 ? -1 : 1;
  const a = Math.abs(n);
  const b = Math.abs(d);
  const g = gcd(a, b) || 1;
  const p = a / g;
  const q = b / g;
  if (q === 1) return `${sign * p}`;
  if (p >= q) {
    throw new Error(
      `가분수 ${sign * p}/${q} — 지면 전처리가 대분수로 고쳐 쓴다(mixed-number-preprocess-20260823.md)`,
    );
  }
  return `${sign < 0 ? "-" : ""}\\dfrac{${p}}{${q}}`;
}

/** `n√r/d` 를 LaTeX 로. 근호가 있으면 분자가 숫자가 아니라 전처리가 안 문다. */
function surdText(s: Surd): string {
  if (s.n === 0) return "0";
  if (s.r === 1) return frac(s.n, s.d);
  const sign = s.n < 0 ? "-" : "";
  const mag = Math.abs(s.n);
  const top = mag === 1 ? `\\sqrt{${s.r}}` : `${mag}\\sqrt{${s.r}}`;
  return s.d === 1 ? `${sign}${top}` : `${sign}\\dfrac{${top}}{${s.d}}`;
}

/** `c0 + c1√root` 를 괄호 없이 — 길이·넓이 답의 표준 꼴. */
function binomText(c0: number, c1: number, root: number): string {
  if (c1 === 0) return `${c0}`;
  const t = surdText(surd(c1, root, 1));
  if (c0 === 0) return t;
  if (c0 > 0) return c1 > 0 ? `${c0}+${t}` : `${c0}${t}`;
  // 유리항이 음수면 무리항을 앞에 둔다 — 교과서 표기는 `4\sqrt{3}-4` 이지
  // `-4+4\sqrt{3}` 이 아니다.
  return `${t}${c0}`;
}

/* ────────────────────────────────────────────────────────────────────────
 * 특수각의 정확한 삼각비 값
 * ──────────────────────────────────────────────────────────────────────── */

const FN_NAMES = ["\\sin", "\\cos", "\\tan"] as const;

/** `[fn][angle]` — `tan 90°` 는 없다(`null`). */
const EXACT: Record<number, (Surd | null)[]> = {
  0: [surd(0, 1, 1), surd(1, 1, 1), surd(0, 1, 1)],
  30: [surd(1, 1, 2), surd(1, 3, 2), surd(1, 3, 3)],
  45: [surd(1, 2, 2), surd(1, 2, 2), surd(1, 1, 1)],
  60: [surd(1, 3, 2), surd(1, 1, 2), surd(1, 3, 1)],
  90: [surd(1, 1, 1), surd(0, 1, 1), null],
};

function exact(fn: number, angle: number): Surd | null {
  const row = EXACT[angle];
  if (!row) throw new Error(`특수각이 아닙니다: ${angle}`);
  return row[fn] ?? null;
}

/** `\sin 30^\circ` 처럼 — 발문·해설이 같은 문자열을 쓰게 한다. */
function trigText(fn: number, angle: number): string {
  return `${FN_NAMES[fn]} ${angle}^\\circ`;
}

const OP_TEXT = ["+", "-", "\\times", "\\div"] as const;

function applyOp(op: number, a: Surd, b: Surd): Surd | null {
  if (op === 0) return surdAdd(a, b);
  if (op === 1) return surdSub(a, b);
  if (op === 2) return surdMul(a, b);
  if (b.n === 0) return null;
  return surdDiv(a, b);
}

 * `lint_svg` 규칙 4 를 정면으로 어긴다.
 *
 * `dims` 의 키는 **두 점 이름**(`"BC"`)이고 엔진 이름은 `d_` 를 붙인다 — 기하 객체
 * 이름은 한 네임스페이스를 쓴다(09 §4-2).
 * ──────────────────────────────────────────────────────────────────────── */

type FigureSpec = Record<string, unknown>;
type XY = [number, number];

const DEG = Math.PI / 180;

/**
 * 수학 좌표(y-위) → SVG 좌표(y-아래), 220×190 균일 배율 (m2GeoSim 과 같은 규약).
 * export 하는 이유: 시험이 캔버스 상수를 옮겨 적지 않게. 문턱은 안 든다.
 */
export function fitPts(pts: Record<string, XY>): Record<string, XY> {
  const xs = Object.values(pts).map((p) => p[0]);
  const ys = Object.values(pts).map((p) => p[1]);
  const minX = Math.min(...xs);
  const maxX = Math.max(...xs);
  const minY = Math.min(...ys);
  const maxY = Math.max(...ys);
  const m = 22;
  const s = Math.min(
    (220 - 2 * m) / Math.max(maxX - minX, 1e-9),
    (190 - 2 * m) / Math.max(maxY - minY, 1e-9),
  );
  const out: Record<string, XY> = {};
  for (const [k, [x, y]] of Object.entries(pts)) {
    out[k] = [r2(m + (x - minX) * s), r2(190 - m - (y - minY) * s)];
  }
  return out;
}

/**
 * 맞춘 뒤 변 길이(px). 치수 곡선을 붙이려면 이 길이는 되어야 한다.
 *
 * 실측(제품 경로, `geometry=1`): 80px(AB=12·BC=16·150°)은 던지고
 * 83px(AB=8·BC=13·120°)은 그린다. 5px 여유.
 *
 * 쪽을 `left`/`right` 로 고정해도 안 된다 — left 는 text, right 는 geometry.
 * 짧은 변에 억지로 달지 말고 값을 다시 뽑는다(m2Pytha 마름모 · 09 §4-27).
 */
const MIN_DIM_SIDE_PX = 85;

function fittedSidePx(pts: Record<string, XY>, p: string, q: string): number {
  const fitted = fitPts(pts);
  const u = fitted[p]!;
  const v = fitted[q]!;
  return Math.hypot(u[0] - v[0], u[1] - v[1]);
}

type Draft = {
  pts: Record<string, XY>;
  segments: Record<string, unknown>;
  angles?: Record<string, unknown>;
  /** 변 이름(두 점) → 그 변에 붙일 치수. 예: `{ AB: "8 cm" }` */
  dims?: Record<string, string>;
  /**
   * 치수 쪽 고정 — `"left"`/`"right"`. 안 주면 `"auto"`(엔진이 둘 다 대 본다).
   * ⚠️ 「도형 안을 지나는 선」(수선 등)에 auto 를 주면 엔진이 안쪽으로도 대 보다가
   *    다른 변과 부딪혀 던질 수 있다 — 그런 변만 쪽을 못 박는다.
   */
  side?: Record<string, "left" | "right">;
  labels: Record<string, string>;
};

/**
 * 좌표를 맞추고 치수를 붙인다.
 *
 * ⚠️ `label` 은 **plain text** 여야 한다 — `dimensions.label` 은 KaTeX 로 다시
 *    렌더링되지 않는다(09 §4-26). `dimCm()` 이 그것을 지킨다.
 */
function build(d: Draft): FigureSpec {
  const fitted = fitPts(d.pts);
  const spec: FigureSpec = {
    version: 2,
    points: { ...fitted },
    segments: d.segments,
    labels: { ...d.labels },
  };
  const dims = Object.entries(d.dims ?? {});
  if (dims.length > 0) {
    const dimensions: Record<string, unknown> = {};
    for (const [pair, text] of dims) {
      const p = pair[0]!;
      const q = pair[1]!;
      if (!fitted[p] || !fitted[q])
        throw new Error(`치수의 끝점이 없습니다: ${pair}`);
      dimensions[`d_${pair}`] = {
        points: [p, q],
        label: text,
        side: d.side?.[pair] ?? "auto",
      };
    }
    spec.dimensions = dimensions;
  }
  if (d.angles && Object.keys(d.angles).length > 0) spec.angles = d.angles;
  return spec;
}

/**
 * 직각삼각형 — $\angle B=90^\circ$, $B$ 는 왼쪽 아래, $A$ 는 위, $C$ 는 오른쪽.
 * 세로·가로에는 **발문의 실제 값**을 넣는다(도형이 값과 안 맞으면 지면에서 보인다).
 */
function figRight(opts: {
  vert: number;
  horz: number;
  dims?: Record<string, string>;
  angA?: string;
  angC?: string;
}): FigureSpec {
  const k = 130 / Math.max(opts.vert, opts.horz);
  const pts: Record<string, XY> = {
    B: [0, 0],
    A: [0, r2(opts.vert * k)],
    C: [r2(opts.horz * k), 0],
  };
  const angles: Record<string, unknown> = {
    r_B: { vertex: "B", points: ["A", "C"], right: true },
  };
  if (opts.angA) {
    angles.a_A = { vertex: "A", points: ["B", "C"], label: opts.angA };
  }
  if (opts.angC) {
    angles.a_C = { vertex: "C", points: ["A", "B"], label: opts.angC };
  }
  return build({
    pts,
    segments: { AB: ["A", "B"], BC: ["B", "C"], CA: ["C", "A"] },
    angles,
    dims: opts.dims,
    labels: { A: "A", B: "B", C: "C" },
  });
}

/**
 * $\angle A=90^\circ$ 인 직각삼각형과 빗변 위의 수선의 발 $H$ (닮음 유형).
 * `ab`·`ac` 는 두 직각변의 실제 길이 — $H$ 자리도 그 값에서 나온다.
 */
function figAltitude(opts: {
  ab: number;
  ac: number;
  dims?: Record<string, string>;
  angBAH?: string;
  angCAH?: string;
}): FigureSpec {
  const a = Math.hypot(opts.ab, opts.ac);
  const k = 150 / a;
  const bh = (opts.ab * opts.ab) / a;
  const ah = (opts.ab * opts.ac) / a;
  const pts: Record<string, XY> = {
    B: [0, 0],
    C: [r2(a * k), 0],
    H: [r2(bh * k), 0],
    A: [r2(bh * k), r2(ah * k)],
  };
  const angles: Record<string, unknown> = {
    r_A: { vertex: "A", points: ["B", "C"], right: true },
    r_H: { vertex: "H", points: ["A", "C"], right: true },
  };
  if (opts.angBAH) {
    angles.a_BAH = { vertex: "A", points: ["B", "H"], label: opts.angBAH };
  }
  if (opts.angCAH) {
    angles.a_CAH = { vertex: "A", points: ["C", "H"], label: opts.angCAH };
  }
  return build({
    pts,
    segments: {
      BC: ["B", "C"],
      AB: ["A", "B"],
      AC: ["A", "C"],
      AH: ["A", "H"],
    },
    angles,
    dims: opts.dims,
    labels: { A: "A", B: "B", C: "C", H: "H" },
  });
}

/** 두 변과 끼인각 — $\angle B$ 가 끼인각. `ab`·`bc` 는 실제 길이. */
function figTwoSides(opts: {
  ab: number;
  bc: number;
  angB: number;
  dims?: Record<string, string>;
}): FigureSpec {
  const k = 140 / Math.max(opts.ab, opts.bc);
  const pts: Record<string, XY> = {
    B: [0, 0],
    C: [r2(opts.bc * k), 0],
    A: [
      r2(opts.ab * k * Math.cos(opts.angB * DEG)),
      r2(opts.ab * k * Math.sin(opts.angB * DEG)),
    ],
  };
  return build({
    pts,
    segments: { AB: ["A", "B"], BC: ["B", "C"], CA: ["C", "A"] },
    angles: {
      a_B: { vertex: "B", points: ["A", "C"], label: `${opts.angB}°` },
    },
    dims: opts.dims,
    labels: { A: "A", B: "B", C: "C" },
  });
}

/**
 * 한 변과 양 끝 각 — `foot` 이면 꼭짓점 $A$ 에서 내린 수선을 **점선**으로 같이 긋는다.
 */
function figTwoAngles(opts: {
  bc: number;
  angB: number;
  angC: number;
  foot?: boolean;
  dims?: Record<string, string>;
}): FigureSpec {
  const cotB = 1 / Math.tan(opts.angB * DEG);
  const cotC = 1 / Math.tan(opts.angC * DEG);
  const h = opts.bc / (cotB + cotC);
  const k = 150 / Math.max(opts.bc, h * 1.4);
  const ax = h * cotB;
  const pts: Record<string, XY> = {
    B: [0, 0],
    C: [r2(opts.bc * k), 0],
    A: [r2(ax * k), r2(h * k)],
  };
  const segments: Record<string, unknown> = {
    AB: ["A", "B"],
    BC: ["B", "C"],
    CA: ["C", "A"],
  };
  const angles: Record<string, unknown> = {
    a_B: { vertex: "B", points: ["A", "C"], label: `${opts.angB}°` },
    a_C: { vertex: "C", points: ["A", "B"], label: `${opts.angC}°` },
  };
  const labels: Record<string, string> = { A: "A", B: "B", C: "C" };
  if (opts.foot) {
    pts.H = [r2(ax * k), 0];
    segments.AH = { points: ["A", "H"], dash: "5 4" };
    angles.r_H = { vertex: "H", points: ["A", "C"], right: true };
    labels.H = "H";
  }
  return build({ pts, segments, angles, dims: opts.dims, labels });
}

/**
 * 한 각이 둔각인 삼각형의 높이 — 수선의 발 $H$ 가 $\overline{BC}$ 의 **바깥**에 선다.
 * `outer` 는 $\angle ABH(=180^\circ-\angle ABC)$, `angC` 는 예각이다.
 *
 * ⚠️ 직각 표시의 반쪽은 $\overline{HB}$ 로 잡는다 — $H$ 에서 $C$ 로 가는 **선분이
 * 없어서** `points:["A","C"]` 로 쓰면 엔진이 「ray is not covered by a rendered
 * segment」로 던진다(실측). $\overline{HB}$ 는 $\overline{HC}$ 와 같은 직선이라
 * 표시하는 각은 같다.
 */
function figObtuseHeight(opts: {
  bc: number;
  outer: number;
  angC: number;
  dims?: Record<string, string>;
}): FigureSpec {
  const cotOuter = 1 / Math.tan(opts.outer * DEG);
  const cotC = 1 / Math.tan(opts.angC * DEG);
  const h = opts.bc / (cotC - cotOuter);
  const back = h * cotOuter;
  const pts: Record<string, XY> = {
    H: [0, 0],
    B: [r2(back), 0],
    C: [r2(back + opts.bc), 0],
    A: [0, r2(h)],
  };
  return build({
    pts,
    segments: {
      AB: ["A", "B"],
      BC: ["B", "C"],
      CA: ["C", "A"],
      AH: { points: ["A", "H"], dash: "5 4" },
      HB: { points: ["H", "B"], dash: "5 4" },
    },
    angles: {
      r_H: { vertex: "H", points: ["A", "B"], right: true },
      a_C: { vertex: "C", points: ["A", "B"], label: `${opts.angC}°` },
    },
    dims: opts.dims,
    labels: { A: "A", B: "B", C: "C", H: "H" },
  });
}

/** 대각선 $AC$ 로 나뉜 사각형 $ABCD$ — 두 삼각형의 끼인각이 $\angle B$·$\angle D$. */
function figQuadDiag(opts: {
  ab: number;
  bc: number;
  angB: number;
  ad: number;
  dc: number;
  angD: number;
  dims?: Record<string, string>;
}): FigureSpec {
  const B: XY = [0, 0];
  const C: XY = [opts.bc, 0];
  const A: XY = [
    opts.ab * Math.cos(opts.angB * DEG),
    opts.ab * Math.sin(opts.angB * DEG),
  ];
  const ac = Math.hypot(C[0] - A[0], C[1] - A[1]);
  // $A$ 에서 $\overline{AC}$ 방향으로 t, 그 수직으로 hd — $B$ 의 반대쪽에 $D$.
  const t = (opts.ad * opts.ad - opts.dc * opts.dc + ac * ac) / (2 * ac);
  const hd = Math.sqrt(Math.max(opts.ad * opts.ad - t * t, 0));
  const ux = (C[0] - A[0]) / ac;
  const uy = (C[1] - A[1]) / ac;
  // 부호 규약: 점 `P` 의 «쪽»은 `(P-A)×u` 의 z 성분이다. 아래 오프셋 `s·(uy,-ux)` 의
  // z 성분은 그대로 `s` 이므로 **`B` 의 반대쪽은 `-sideB`** 다.
  // (2026-08-23: 처음엔 `+sideB` 라 `D` 가 삼각형 안으로 들어와 사각형이 겹쳤다.
  //  발문은 사각형이라 하는데 그림은 아니었다 — 미리보기에서 드러났다.)
  const sideB = Math.sign((B[0] - A[0]) * uy - (B[1] - A[1]) * ux) || 1;
  const s = -sideB;
  const D: XY = [
    r2(A[0] + ux * t + s * uy * hd),
    r2(A[1] + uy * t - s * ux * hd),
  ];
  return build({
    pts: { A: [r2(A[0]), r2(A[1])], B, C, D },
    segments: {
      AB: ["A", "B"],
      BC: ["B", "C"],
      CD: ["C", "D"],
      DA: ["D", "A"],
      AC: ["A", "C"],
    },
    angles: {
      a_B: { vertex: "B", points: ["A", "C"], label: `${opts.angB}°` },
      a_D: { vertex: "D", points: ["A", "C"], label: `${opts.angD}°` },
    },
    dims: opts.dims,
    labels: { A: "A", B: "B", C: "C", D: "D" },
  });
}

/** 평행사변형 $ABCD$ — $\angle A$ 가 이웃한 두 변 $\overline{AB}$·$\overline{AD}$ 의 끼인각. */
function figParallelogram(opts: {
  ab: number;
  ad: number;
  angA: number;
  dims?: Record<string, string>;
}): FigureSpec {
  const k = 140 / Math.max(opts.ab, opts.ad);
  const dx = opts.ad * k * Math.cos(opts.angA * DEG);
  const dy = opts.ad * k * Math.sin(opts.angA * DEG);
  const pts: Record<string, XY> = {
    A: [0, 0],
    B: [r2(opts.ab * k), 0],
    C: [r2(opts.ab * k + dx), r2(dy)],
    D: [r2(dx), r2(dy)],
  };
  return build({
    pts,
    segments: {
      AB: ["A", "B"],
      BC: ["B", "C"],
      CD: ["C", "D"],
      DA: ["D", "A"],
    },
    angles: {
      a_A: { vertex: "A", points: ["B", "D"], label: `${opts.angA}°` },
    },
    dims: opts.dims,
    labels: { A: "A", B: "B", C: "C", D: "D" },
  });
}

/**
 * 두 대각선이 점 $P$ 에서 각 `ang` 로 만나는 사각형 $ABCD$.
 *
 * ⚠️ 대각선은 $\overline{AP}$·$\overline{PC}$·$\overline{BP}$·$\overline{PD}$ 넷으로
 * **쪼개서** 긋는다 — 통째로 그으면 $P$ 가 끝점이 아니라서 $\angle DPC$ 표시가
 * 「ray is not covered by a rendered segment」로 거부된다(실측).
 */
function figQuadCross(opts: {
  d1: number;
  d2: number;
  ang: number;
  dims?: Record<string, string>;
}): FigureSpec {
  const k = 150 / Math.max(opts.d1, opts.d2);
  const u = 0.45 * opts.d1 * k;
  const v = 0.4 * opts.d2 * k;
  const cx = Math.cos(opts.ang * DEG);
  const cy = Math.sin(opts.ang * DEG);
  const pts: Record<string, XY> = {
    P: [0, 0],
    A: [r2(-u), 0],
    C: [r2(opts.d1 * k - u), 0],
    B: [r2(-v * cx), r2(-v * cy)],
    D: [r2((opts.d2 * k - v) * cx), r2((opts.d2 * k - v) * cy)],
  };
  return build({
    pts,
    segments: {
      AB: ["A", "B"],
      BC: ["B", "C"],
      CD: ["C", "D"],
      DA: ["D", "A"],
      AP: ["A", "P"],
      PC: ["P", "C"],
      BP: ["B", "P"],
      PD: ["P", "D"],
    },
    angles: {
      a_P: { vertex: "P", points: ["C", "D"], label: `${opts.ang}°` },
    },
    dims: opts.dims,
    labels: { A: "A", B: "B", C: "C", D: "D", P: "P" },
  });
}

const CX = 118;
const CY = 100;
const R_PX = 82;

/**
 * 반지름 $1$ 인 사분원 — $\angle AOB=$`ang`, $\overline{BH}\perp\overline{OA}$,
 * $\overline{AT}\perp\overline{OA}$. 원은 **호만** 그린다(`draw:false` + `arcs`).
 *
 * ⚠️ $\overline{OT}$ 를 통째로 긋지 않고 $\overline{OB}$·$\overline{BT}$ 로 나눈다 —
 * 그래야 $\angle AOB$ 표시의 $B$ 쪽 반쪽이 «그려진 선분»에 얹힌다(실측).
 */
function figQuadrant(opts: { ang: number }): FigureSpec {
  const t = Math.tan(opts.ang * DEG);
  const rad = t > 1.35 ? 62 : R_PX;
  const bx = rad * Math.cos(opts.ang * DEG);
  return {
    version: 2,
    points: {
      O: [CX, CY],
      A: { type: "on_circle", circle: "c", angle: 0 },
      Q: { type: "on_circle", circle: "c", angle: 90 },
      B: { type: "on_circle", circle: "c", angle: opts.ang },
      H: [r2(CX + bx), CY],
      T: [r2(CX + rad), r2(CY - rad * t)],
    },
    circles: { c: { center: "O", radius: rad, draw: false } },
    arcs: { q_AQ: { circle: "c", from: "A", to: "Q", dir: "ccw" } },
    segments: {
      OA: ["O", "A"],
      OQ: ["O", "Q"],
      OB: ["O", "B"],
      BT: ["B", "T"],
      BH: ["B", "H"],
      AT: ["A", "T"],
    },
    angles: {
      a_O: { vertex: "O", points: ["A", "B"], label: `${opts.ang}°` },
      r_H: { vertex: "H", points: ["B", "O"], right: true },
      r_A: { vertex: "A", points: ["O", "T"], right: true },
    },
    labels: { O: "O", A: "A", B: "B", H: "H", T: "T" },
  };
}

/**
 * 반지름 `R` 인 원에 내접하는 정 `n` 각형 — 중심 $O$ 와 반지름 하나를 긋는다.
 *
 * ⚠️ **반지름에 값은 안 적는다.** 두 가지가 겹친다.
 *   ⑴ 발문이 이미 「반지름의 길이가 $R$ cm 인 원」이라 말한다 — 09 §4-20-b 의
 *      「발문이 이미 말했는가」(원장님 확정 2026-08-22)에 그대로 해당한다.
 *   ⑵ 그리고 **들어갈 자리가 없다.** 정 $n$ 각형 안에서 반지름 옆 통로는 대략
 *      $R/2$ 인데 값 상자는 45단위쯤이다. 실측(반지름 82~120 × $n$=3·4·6·8·12):
 *      **정삼각형은 어느 크기에서도 안 들어가고** 정사각형은 반지름 95부터 들어간다.
 *      그렇다고 날 `<text>` 를 옆에 놓으면 그것은 「선에서 떨어진 맨 길이 표기」라
 *      lint 규칙 4 위반이다 — 종전 `n_R` 이 바로 그것이었고, 이번 판정기가 물었다.
 */
function figRegularPoly(opts: { n: number }): FigureSpec {
  const names = Array.from({ length: opts.n }, (_, i) => `P${i + 1}`);
  const points: Record<string, unknown> = { O: [CX, CY] };
  const segments: Record<string, unknown> = {};
  const step = 360 / opts.n;
  for (let i = 0; i < opts.n; i += 1) {
    points[names[i]!] = {
      type: "on_circle",
      circle: "c",
      angle: r2(90 + i * step),
    };
    segments[`s${i + 1}`] = [names[i]!, names[(i + 1) % opts.n]!];
  }
  segments.r1 = ["O", names[0]!];
  return {
    version: 2,
    points,
    circles: { c: { center: "O", radius: R_PX } },
    segments,
    labels: { O: "O" },
  };
}

/**
 * 두 지점에서 올려본각을 잰 장면 — 건물 밑면 $H$, 꼭대기 $T$, 먼 지점 $A$, 가까운 지점 $B$.
 * 수선의 발이 $\overline{AB}$ 바깥에 있는 «높이 구하기»와 같은 모양이다.
 */
function figElevationTwo(opts: {
  d: number;
  ang1: number;
  ang2: number;
  label: string;
}): FigureSpec {
  const h =
    opts.d / (1 / Math.tan(opts.ang1 * DEG) - 1 / Math.tan(opts.ang2 * DEG));
  const b1 = h / Math.tan(opts.ang2 * DEG);
  return build({
    pts: {
      H: [0, 0],
      B: [r2(b1), 0],
      A: [r2(b1 + opts.d), 0],
      T: [0, r2(h)],
    },
    segments: {
      HT: ["H", "T"],
      HA: ["H", "A"],
      TB: ["T", "B"],
      TA: ["T", "A"],
    },
    angles: {
      r_H: { vertex: "H", points: ["T", "A"], right: true },
      a_A: { vertex: "A", points: ["T", "H"], label: `${opts.ang1}°` },
      a_B: { vertex: "B", points: ["T", "H"], label: `${opts.ang2}°` },
    },
    dims: { BA: opts.label },
    labels: { A: "A", B: "B", H: "H", T: "T" },
  });
}

/* ────────────────────────────────────────────────────────────────────────
 * 발문 조각 — 같은 문자열을 발문·해설·그림이 함께 쓴다
 * ──────────────────────────────────────────────────────────────────────── */

/** `$` 안에 들어갈 `12\,\mathrm{cm}`. 단위는 언제나 수식 안이다. */
function cm(v: number | string): string {
  return `${v}\\,\\mathrm{cm}`;
}

/** `$` 안에 들어갈 `12\,\mathrm{cm}^2`. */
function cm2(v: number | string): string {
  return `${v}\\,\\mathrm{cm}^2`;
}

/**
 * `facts` 한 칸 — 갈래마다 키가 달라도 **여기서** `Record<string, number>` 로 좁힌다.
 * 안 좁히면 두 갈래의 객체 타입이 합쳐지며 `a?: undefined` 가 생겨 `Body` 에 안 맞는다.
 */
function nums(o: Record<string, number>): Record<string, number> {
  return o;
}

/** 기약분수의 분자·분모. */
function red(n: number, d: number): [number, number] {
  const g = gcd(Math.abs(n), Math.abs(d)) || 1;
  return [n / g, d / g];
}

/** `\sin A`·`\cos A`·`\tan A` (일반각 표기). */
function trigOf(fn: number, name: string): string {
  return `${FN_NAMES[fn]} ${name}`;
}

/**
 * 삼각비의 표 한 칸 — **계산해서** 만든다. 손으로 옮긴 표는 샌다
 * (CLAUDE.md 2026-08-18 「목록을 손으로 쓰면 세는 쪽과 고치는 쪽이 같이 눈이 먼다」).
 * 시험도 같은 방법으로 다시 계산해 대조하므로, 값이 틀리면 시험이 빨개진다.
 */
function tableValue(fn: number, deg: number): number {
  const f = [Math.sin, Math.cos, Math.tan][fn]!;
  return Number(f(deg * DEG).toFixed(4));
}

/** `$(12-4\sqrt{3})\,\mathrm{cm}$` — 두 항이면 괄호로 묶는다(단위가 무리항에만 붙어 보이지 않게). */
function lenText(c0: number, c1: number, root: number, unit: string): string {
  const t = binomText(c0, c1, root);
  const two = c0 !== 0 && c1 !== 0;
  return `${two ? `(${t})` : t}\\,\\mathrm{${unit}}`;
}

/**
 * 삼각형의 넓이 유형(활용 08 예각 · 09 둔각)이 **같은 자를 쓴다**.
 * 둔각은 $\sin(180^\circ-A)$ 로 예각의 값을 그대로 쓴다 — 두 유형이 규칙을
 * 따로 들고 있으면 한쪽만 고쳐진다(이 저장소의 반복 결함).
 */
function triangleArea(rng: Rng, angles: readonly number[]): Body {
  return attempt(200, () => {
    const ang = pick(rng, angles);
    const a = intBetween(rng, 3, 16);
    const b = intBetween(rng, 3, 16);
    const sinT = exact(0, ang > 90 ? 180 - ang : ang)!;
    const area = surdMul(surdMul(surd(a * b, 1, 1), sinT), surd(1, 1, 2));
    if (area.d !== 1) return null;
    // 짧은 $\overline{AB}$ 에 치수를 달면 엔진이 변과 겹쳐 던진다
    // (J30503-A09, 적재 씨앗 20260827 v0: 4cm/16cm @120° → 맞춘 뒤 39px).
    const ax = a * Math.cos(ang * DEG);
    const raw: Record<string, XY> = {
      B: [0, 0],
      C: [b, 0],
      A: [ax, a * Math.sin(ang * DEG)],
    };
    if (fittedSidePx(raw, "A", "B") < MIN_DIM_SIDE_PX) return null;
    // 끼인각 B 가 예각이어도 $a\cos B > b$ 이면 C 가 둔각이라 C 가 AB 밑에

    // (활용 08, 15cm/8cm @30° — 맞춘 뒤 AB 는 203px 로 **긴데** 던진다).
    if (ax > b + 1e-6) return null;
    const mode = intBetween(rng, 0, 1);
    const shrink =
      ang > 90 ? `\\sin(180^\\circ-${ang}^\\circ)` : `\\sin ${ang}^\\circ`;

    if (mode === 0) {
      return {
        content: `삼각형 $ABC$에서 $\\overline{AB}=${cm(a)}$, $\\overline{BC}=${cm(b)}$, $\\angle B=${ang}^\\circ$일 때, 삼각형 $ABC$의 넓이를 구하시오.`,
        answer: `$${cm2(surdText(area))}$`,
        solution: `${rule}이다.\n따라서 넓이는 $\\dfrac{1}{2}\\times${a}\\times${b}\\times${surdText(sinT)}=${surdText(area)}\\,(\\mathrm{cm}^2)$이다.`,
        facts: { a, b, ang, mode, an: area.n, ar: area.r },
        figureSpec: figTwoSides({
          ab: a,
          bc: b,
          angB: ang,
          dims: { AB: dimCm(a), BC: dimCm(b) },
        }),
      };
    }
    return {
      content: `삼각형 $ABC$에서 $\\overline{AB}=${cm(a)}$, $\\angle B=${ang}^\\circ$이고 넓이가 $${cm2(surdText(area))}$일 때, $\\overline{BC}$의 길이를 구하시오.`,
      answer: `$${cm(b)}$`,
      solution: `${rule}이다.\n$\\dfrac{1}{2}\\times${a}\\times\\overline{BC}\\times${surdText(sinT)}=${surdText(area)}$이므로 $\\overline{BC}=${b}\\,(\\mathrm{cm})$이다.`,
      facts: { a, b, ang, mode, an: area.n, ar: area.r },
      figureSpec: figTwoSides({
        ab: a,
        bc: b,
        angB: ang,
        dims: { AB: dimCm(a) },
      }),
    };
  });
}
