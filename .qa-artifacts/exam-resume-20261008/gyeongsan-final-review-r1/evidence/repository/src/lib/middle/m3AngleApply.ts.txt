 * 있으므로 어느 것을 물어도 같은 그림이 답한다. 그림의 점은 전부 **발문의 실제 각**

 *   작도원으로 점만 얹고, 07 은 원 자체가 없다.
 * - 각 이름은 `ang…`, 점 이름과 겹치지 않게 한다(09 §4-2).

const DEG = Math.PI / 180;

/** 조건을 만족할 때까지 다시 뽑는다 — 못 찾으면 던진다(조용히 기본값 금지). */
function attempt<T>(tries: number, f: () => T | null): T {
  for (let i = 0; i < tries; i += 1) {
    const v = f();
    if (v !== null) return v;
  }
  throw new Error("유형 생성 재시도 초과 — 매개변수 범위를 확인하십시오");
}

type Pt = readonly [number, number];
type FigureSpec = Record<string, unknown>;
type SpecPoint = Pt | Record<string, unknown>;

/**
 * 스펙에 실을 각(도) — **여기서 눕힌 값으로 좌표도 계산한다.** 엔진과 우리가 같은
 * 각을 쓰지 않으면 교점·수선의 발이 미세하게 어긋난다. 넷째 자리까지 남기는 이유는
 * 시험이 그림에서 **길이비**까지 재기 때문이다(둘째 자리면 상대오차 1e-4 가 남는다).
 */
function onAng(deg: number): number {
  return Math.round((((deg % 360) + 360) % 360) * 1e4) / 1e4;
}

/** 수학각(도) → SVG 좌표 (y-down 보정). 엔진 `circle_pt` 와 같은 규약. */
function polar(c: Pt, angleDeg: number, radius: number): Pt {
  return [
    c[0] + radius * Math.cos(angleDeg * DEG),
    c[1] - radius * Math.sin(angleDeg * DEG),
  ];
}

function sub(a: Pt, b: Pt): Pt {
  return [a[0] - b[0], a[1] - b[1]];
}

function add(a: Pt, b: Pt): Pt {
  return [a[0] + b[0], a[1] + b[1]];
}

function mul(a: Pt, k: number): Pt {
  return [a[0] * k, a[1] * k];
}

function unit(v: Pt): Pt {
  const n = Math.hypot(v[0], v[1]);
  if (n < 1e-12) throw new Error("그림 계산 실패: 길이가 0 인 방향벡터");
  return [v[0] / n, v[1] / n];
}

/** p1 + t·d1 == p2 + s·d2 의 교점. 평행이면 던진다. 반올림 없음. */
function lineCross(p1: Pt, d1: Pt, p2: Pt, d2: Pt): Pt {
  const det = d1[0] * d2[1] - d1[1] * d2[0];
  if (Math.abs(det) < 1e-9) {
    throw new Error("그림 계산 실패: 두 직선이 평행합니다");
  }
  const t = ((p2[0] - p1[0]) * d2[1] - (p2[1] - p1[1]) * d2[0]) / det;
  return [p1[0] + t * d1[0], p1[1] + t * d1[1]];
}

/** 점 o 에서 직선 ab 에 내린 수선의 발. 반올림하지 않는다(직각 마크 허용오차). */
function foot(o: Pt, a: Pt, b: Pt): Pt {
  const ux = b[0] - a[0];
  const uy = b[1] - a[1];
  const t = ((o[0] - a[0]) * ux + (o[1] - a[1]) * uy) / (ux * ux + uy * uy);
  return [a[0] + t * ux, a[1] + t * uy];
}

/** 중심 c·반지름 r 인 원 위의 점 t 에서 방향 u 로 그은 현의 반대 끝점. */
function chordOther(c: Pt, t: Pt, u: Pt): Pt {
  const d = sub(c, t);
  return add(t, mul(u, 2 * (d[0] * u[0] + d[1] * u[1])));
}

/**
 * 지면 등급은 viewBox **폭 하나**로 갈린다(`figureSvgFrame`: ≤160 → 140px, ≤320 → 240px).
 * 폭을 안 박으면 그 값이 **각도에 따라 떠다녀** 같은 유형인데 지면 글자 크기가 달라진다
 * (CLAUDE.md 2026-08-23 · 09 §4-24). 실측 2026-08-24: 안 박았을 때 158~357 이었다.
 *
 * 그래서 그림을 다 짓고 **마지막에 통째로 늘여** 내용 폭을 못 박는다. 원 위의 점은
 * (원, 각)으로 정의돼 있으므로 원의 중심·반지름만 같이 늘이면 그대로 따라온다.
 * 닮음변환이라 각·직각 표시·수선의 발은 그대로다.
 */
const FIT_WIDTH = 188;
const FIT_MARGIN = 6;

/** 스펙의 점을 좌표로 되살린다 — 엔진과 같은 규약(y-down, 수학각). */
function resolveAll(
  points: Record<string, SpecPoint>,
  circles: Record<string, { center: string; radius: number; draw?: boolean }>,
  segments: Record<string, unknown>,
): Record<string, Pt> {
  const out: Record<string, Pt> = {};
  const ends = (v: unknown): [string, string] =>
    Array.isArray(v)
      ? [v[0] as string, v[1] as string]
      : (v as { points: [string, string] }).points;
  for (const [n, v] of Object.entries(points)) {
    if (Array.isArray(v)) out[n] = [v[0] as number, v[1] as number];
  }
  for (const [n, v] of Object.entries(points)) {
    const o = v as { type?: string; circle?: string; angle?: number };
    if (o?.type !== "on_circle") continue;
    const c = circles[o.circle!]!;
    out[n] = polar(out[c.center]!, o.angle!, c.radius);
  }
  for (const [n, v] of Object.entries(points)) {
    const o = v as { type?: string; segments?: [string, string] };
    if (o?.type !== "intersection") continue;
    const [s1, s2] = o.segments!;
    const [a, b] = ends(segments[s1]);
    const [c, d] = ends(segments[s2]);
    out[n] = lineCross(
      out[a]!,
      sub(out[b]!, out[a]!),
      out[c]!,
      sub(out[d]!, out[c]!),
    );
  }
  return out;
}

function spec(
  points: Record<string, SpecPoint>,
  circles: Record<string, unknown>,
  rest: Record<string, unknown>,
  showPoints = true,
): FigureSpec {
  const circ = circles as Record<
    string,
    { center: string; radius: number; draw?: boolean }
  >;
  const segments = (rest.segments ?? {}) as Record<string, unknown>;
  const resolved = resolveAll(points, circ, segments);
  let minX = Infinity;
  let maxX = -Infinity;
  let minY = Infinity;
  const touch = (x: number, y: number): void => {
    minX = Math.min(minX, x);
    maxX = Math.max(maxX, x);
    minY = Math.min(minY, y);
  };
  for (const p of Object.values(resolved)) touch(p[0], p[1]);
  for (const c of Object.values(circ)) {
    if (c.draw === false) continue;
    const ctr = resolved[c.center]!;
    touch(ctr[0] - c.radius, ctr[1] - c.radius);
    touch(ctr[0] + c.radius, ctr[1] + c.radius);
  }
  const width = maxX - minX;
  if (!(width > 1e-6)) throw new Error("그림 계산 실패: 폭이 0 입니다");
  const s = FIT_WIDTH / width;
  const map = (p: Pt): Pt => [
    (p[0] - minX) * s + FIT_MARGIN,
    (p[1] - minY) * s + FIT_MARGIN,
  ];
  const outPoints: Record<string, SpecPoint> = {};
  for (const [n, v] of Object.entries(points)) {
    outPoints[n] = Array.isArray(v) ? map([v[0] as number, v[1] as number]) : v;
  }
  const outCircles: Record<string, unknown> = {};
  for (const [n, c] of Object.entries(circ)) {
    outCircles[n] = { ...c, radius: c.radius * s };
  }
  return {
    version: 2,
    style: { show_points: showPoints },
    points: outPoints,
    circles: outCircles,
    ...rest,
  };
}

function deg(v: number): string {
  return `${v}°`;
}

/**
 * 01 네 점이 한 원 위에 있을 조건 — **원은 그리지 않는다**(작도원 `draw:false`).
 * 호: AB=p, BC=2a, CD=p, DA=2c (a+p+c=180 이라 합이 360).
 */
function figConcyclic(o: {
  a: number;
  p: number;
  c: number;
  labA: string;
  labP: string;
  labC: string;
}): FigureSpec {
  const R = 74;
  const O: Pt = [118, 100];
  const t0 = 145;
  const tA = onAng(t0);
  const tB = onAng(t0 - o.p);
  const tC = onAng(t0 - o.p - 2 * o.a);
  const tD = onAng(t0 - 2 * o.p - 2 * o.a);
  return spec(
    {
      O,
      A: { type: "on_circle", circle: "k", angle: tA },
      B: { type: "on_circle", circle: "k", angle: tB },
      C: { type: "on_circle", circle: "k", angle: tC },
      D: { type: "on_circle", circle: "k", angle: tD },
      P: { type: "intersection", segments: ["AC", "BD"] },
    },
    { k: { center: "O", radius: R, draw: false } },
    {
      segments: {
        AB: ["A", "B"],
        BC: ["B", "C"],
        CD: ["C", "D"],
        DA: ["D", "A"],
        AC: ["A", "C"],
        BD: ["B", "D"],
      },
      angles: {
        angA: { vertex: "A", points: ["B", "C"], label: o.labA },
        angP: { vertex: "P", points: ["A", "B"], label: o.labP },
        angC: { vertex: "C", points: ["A", "D"], label: o.labC },
      },
      labels: { A: "A", B: "B", C: "C", D: "D", P: "P" },
    },
  );
}

/** 원에 내접하는 사각형 ABCD (02·03·09 의 뼈대). 각·추가 선분은 호출자가 준다. */
function cyclicQuadSpec(
  thetas: [number, number, number, number],
  opts: {
    R?: number;
    center?: Pt;
    extra?: Record<string, unknown>;
    extraPoints?: Record<string, SpecPoint>;
    segments: Record<string, unknown>;
    angles: Record<string, unknown>;
    labels: Record<string, string>;
  },
): FigureSpec {
  const R = opts.R ?? 72;
  const O: Pt = opts.center ?? [116, 98];
  const [tA, tB, tC, tD] = thetas;
  return spec(
    {
      O,
      A: { type: "on_circle", circle: "k", angle: onAng(tA) },
      B: { type: "on_circle", circle: "k", angle: onAng(tB) },
      C: { type: "on_circle", circle: "k", angle: onAng(tC) },
      D: { type: "on_circle", circle: "k", angle: onAng(tD) },
      ...(opts.extraPoints ?? {}),
    },
    { k: { center: "O", radius: R } },
    {
      segments: opts.segments,
      angles: opts.angles,
      labels: opts.labels,
      ...(opts.extra ?? {}),
    },
  );
}

/** 02 원에 내접하는 사각형 — 대각의 합. mode 2 만 대각선 AC 를 그린다. */
function figCyclicQuad(o: {
  thetas: [number, number, number, number];
  diag: boolean;
  angles: Record<string, unknown>;
}): FigureSpec {
  const segments: Record<string, unknown> = {
    AB: ["A", "B"],
    BC: ["B", "C"],
    CD: ["C", "D"],
    DA: ["D", "A"],
  };
  if (o.diag) segments.AC = ["A", "C"];
  return cyclicQuadSpec(o.thetas, {
    segments,
    angles: o.angles,
    labels: { A: "A", B: "B", C: "C", D: "D" },
  });
}

/** 03 외각 — 변 BC 의 연장선 위의 점 E. mode 2 만 대각선 BD 를 그린다. */
function figCyclicQuadExt(o: {
  thetas: [number, number, number, number];
  diagBD: boolean;
  angles: Record<string, unknown>;
}): FigureSpec {
  const R = 68;
  const O: Pt = [104, 98];
  const B = polar(O, onAng(o.thetas[1]), R);
  const C = polar(O, onAng(o.thetas[2]), R);
  const E = add(B, mul(sub(C, B), 1.55));
  const segments: Record<string, unknown> = {
    AB: ["A", "B"],
    BE: ["B", "E"],
    CD: ["C", "D"],
    DA: ["D", "A"],
  };
  if (o.diagBD) segments.BD = ["B", "D"];
  return cyclicQuadSpec(o.thetas, {
    R,
    center: O,
    extraPoints: { E },
    segments,
    angles: o.angles,
    labels: { A: "A", B: "B", C: "C", D: "D", E: "E" },
  });
}

/** 04 원에 내접하는 다각형 — 호 목록으로 꼭짓점을 놓는다. */
function figCyclicPoly(o: {
  names: string[];
  arcs: number[];
  start: number;
  radii: string[];
  angles: Record<string, unknown>;
}): FigureSpec {
  const R = 70;
  const O: Pt = [112, 96];
  const points: Record<string, SpecPoint> = { O };
  const labels: Record<string, string> = {};
  let t = o.start;
  const thetas: number[] = [];
  for (let i = 0; i < o.names.length; i += 1) {
    thetas.push(t);
    points[o.names[i]!] = {
      type: "on_circle",
      circle: "k",
      angle: onAng(t),
    };
    labels[o.names[i]!] = o.names[i]!;
    t -= o.arcs[i]!;
  }
  const segments: Record<string, unknown> = {};
  for (let i = 0; i < o.names.length; i += 1) {
    const p = o.names[i]!;
    const q = o.names[(i + 1) % o.names.length]!;
    segments[`s${p}${q}`] = [p, q];
  }
  for (const n of o.radii) segments[`O${n}`] = ["O", n];
  if (o.radii.length > 0) labels.O = "O";
  return spec(
    points,
    { k: { center: "O", radius: R } },
    {
      segments,
      angles: o.angles,
      labels,
    },
  );
}

/**
 * 05 내접 사각형의 응용 — AB 와 DC 의 연장선이 P 에서, AD 와 BC 의 연장선이 Q 에서 만난다.
 * 네 변은 AP·DP·AQ·BQ 안에 들어 있다(따로 긋지 않는다).
 */
function figCyclicPQ(o: {
  thetas: [number, number, number, number];
  angles: Record<string, unknown>;
}): FigureSpec {
  // 폭이 160 아래로 내려가면 지면 등급이 140px 로 떨어진다(09 §4-24) — R 을 키워
  // 이 유형의 최소 폭을 160 위로 올린다(실측 2026-08-24: R=58 에서 155.0).
  const R = 69;
  const O: Pt = [110, 106];
  const [tA, tB, tC, tD] = o.thetas.map(onAng) as [
    number,
    number,
    number,
    number,
  ];
  const A = polar(O, tA, R);
  const B = polar(O, tB, R);
  const C = polar(O, tC, R);
  const D = polar(O, tD, R);
  const P = lineCross(A, sub(B, A), D, sub(C, D));
  const Q = lineCross(A, sub(D, A), B, sub(C, B));
  return spec(
    {
      O,
      A: { type: "on_circle", circle: "k", angle: tA },
      B: { type: "on_circle", circle: "k", angle: tB },
      C: { type: "on_circle", circle: "k", angle: tC },
      D: { type: "on_circle", circle: "k", angle: tD },
      P,
      Q,
    },
    { k: { center: "O", radius: R } },
    {
      segments: {
        AP: ["A", "P"],
        DP: ["D", "P"],
        AQ: ["A", "Q"],
        BQ: ["B", "Q"],
      },
      angles: o.angles,
      labels: { A: "A", B: "B", C: "C", D: "D", P: "P", Q: "Q" },
    },
  );
}

/**
 * 06 두 원에서 내접하는 사각형의 응용 — 두 원이 P, Q 에서 만나고
 * 세 점 A·P·D 와 세 점 B·Q·C 가 각각 한 직선 위에 있다.
 */
function figTwoCirclesCross(o: {
  a: number;
  withCenter: boolean;
  angles: Record<string, unknown>;
}): FigureSpec {
  const r1 = 56;
  const r2 = 50;
  const O1: Pt = [82, 96];
  const O2: Pt = [82 + 78, 96];
  const d = O2[0] - O1[0];
  const ax = (d * d + r1 * r1 - r2 * r2) / (2 * d);
  const hh = Math.sqrt(r1 * r1 - ax * ax);
  // P 는 위, Q 는 아래 — 원 1 기준의 각으로 되돌려 on_circle 로 얹는다.
  const tP = onAng(Math.atan2(hh, ax) / DEG);
  const tQ = onAng(-Math.atan2(hh, ax) / DEG);
  const tA = onAng(150);
  const tB = onAng(tP - 2 * o.a);
  const P = polar(O1, tP, r1);
  const Q = polar(O1, tQ, r1);
  const A = polar(O1, tA, r1);
  const B = polar(O1, tB, r1);
  const D = chordOther(O2, P, unit(sub(P, A)));
  const C = chordOther(O2, Q, unit(sub(Q, B)));
  const segments: Record<string, unknown> = {
    AD: ["A", "D"],
    BC: ["B", "C"],
    AB: ["A", "B"],
    DC: ["D", "C"],
    PQ: ["P", "Q"],
  };
  // 발문이 「두 점 $A$, $B$는 원 $O$ 위의 점」이라 부르므로 O 라벨은 늘 적는다.
  const labels: Record<string, string> = {
    A: "A",
    B: "B",
    C: "C",
    D: "D",
    P: "P",
    Q: "Q",
    O: "O",
  };
  if (o.withCenter) {
    segments.OP = ["O", "P"];
    segments.OB = ["O", "B"];
  }
  return spec(
    {
      O: O1,
      W: O2,
      P: { type: "on_circle", circle: "k", angle: tP },
      Q: { type: "on_circle", circle: "k", angle: tQ },
      A: { type: "on_circle", circle: "k", angle: tA },
      B: { type: "on_circle", circle: "k", angle: tB },
      C,
      D,
    },
    {
      k: { center: "O", radius: r1 },
      k2: { center: "W", radius: r2 },
    },
    { segments, angles: o.angles, labels },
  );
}

/**
 * 07 사각형이 원에 내접하기 위한 조건 — △ABC 와 변 위의 D·E.
 * **원을 그리지 않는다**(내접하는지가 묻는 것이다).
 */
function figTriangleInner(o: {
  a: number;
  b: number;
  angles: Record<string, unknown>;
}): FigureSpec {
  const B: Pt = [26, 150];
  const C: Pt = [186, 150];
  // ∠B 와 ∠A 로 A 를 잡는다: A 는 B 에서 각 b 로 올라간 직선과 C 에서 각 (180−a−b) 로
  // 올라간 직선의 교점이다.
  const c2 = 180 - o.a - o.b;
  const A = lineCross(B, [Math.cos(-o.b * DEG), Math.sin(-o.b * DEG)], C, [
    -Math.cos(-c2 * DEG),
    Math.sin(-c2 * DEG),
  ]);
  const ab = Math.hypot(A[0] - B[0], A[1] - B[1]);
  const ac = Math.hypot(A[0] - C[0], A[1] - C[1]);
  // △ADE ∽ △ACB 가 되도록 AD:AC = AE:AB = t — 그래야 ∠ADE = ∠ACB 다.
  const t = 0.62 * Math.min(ab / ac, ac / ab);
  const D = add(A, mul(sub(B, A), (t * ac) / ab));
  const E = add(A, mul(sub(C, A), (t * ab) / ac));
  return spec(
    { A, B, C, D, E },
    {},
    {
      segments: {
        AB: ["A", "B"],
        BC: ["B", "C"],
        CA: ["C", "A"],
        DE: ["D", "E"],
      },
      angles: o.angles,
      labels: { A: "A", B: "B", C: "C", D: "D", E: "E" },
    },
  );
}

/**
 * 08 접선과 현이 이루는 각 ⑴ — 접점 B 에서 접선 TT′, 현 BA.
 * `withCenter` 면 반지름 OA·OB 를, `third` 면 세 번째 점 C 를 얹는다.
 */
function figTangentChord(o: {
  a: number;
  arcBC?: number;
  withCenter: boolean;
  angles: Record<string, unknown>;
}): FigureSpec {
  const R = 78;
  const O: Pt = [122, 96];
  const tB = onAng(270);
  const B = polar(O, tB, R);
  const T = [B[0] - 72, B[1]] as Pt;
  const T2 = [B[0] + 72, B[1]] as Pt;
  const tA = onAng(270 - 2 * o.a);
  const points: Record<string, SpecPoint> = {
    O,
    B: { type: "on_circle", circle: "k", angle: tB },
    A: { type: "on_circle", circle: "k", angle: tA },
    T,
    U: T2,
  };
  const segments: Record<string, unknown> = {
    TU: ["T", "U"],
    AB: ["A", "B"],
  };
  const labels: Record<string, string> = { A: "A", B: "B", T: "T", U: "T'" };
  if (o.withCenter) {
    segments.OA = ["O", "A"];
    segments.OB = ["O", "B"];
    labels.O = "O";
  }
  if (o.arcBC !== undefined) {
    points.C = { type: "on_circle", circle: "k", angle: onAng(270 + o.arcBC) };
    segments.BC = ["B", "C"];
    segments.CA = ["C", "A"];
    labels.C = "C";
  }
  return spec(
    points,
    { k: { center: "O", radius: R } },
    {
      segments,
      angles: o.angles,
      labels,
    },
  );
}

/** 08 mode 1 — 접점 T 의 접선이 현 AB 의 연장선과 점 P 에서 만난다. */
function figTangentSecant(o: {
  b: number;
  psi: number;
  angles: Record<string, unknown>;
}): FigureSpec {
  const R = 74;
  const O: Pt = [116, 94];
  const tT = onAng(270);
  const T = polar(O, tT, R);
  const tA = onAng(270 - 2 * o.b);
  const tB = onAng(270 + 2 * o.psi);
  const A = polar(O, tA, R);
  const B = polar(O, tB, R);
  const P = lineCross(A, sub(B, A), T, [1, 0]);
  return spec(
    {
      O,
      T: { type: "on_circle", circle: "k", angle: tT },
      A: { type: "on_circle", circle: "k", angle: tA },
      B: { type: "on_circle", circle: "k", angle: tB },
      P,
    },
    { k: { center: "O", radius: R } },
    {
      segments: {
        AP: ["A", "P"],
        TP: ["T", "P"],
        AT: ["A", "T"],
        BT: ["B", "T"],
      },
      angles: o.angles,
      labels: { A: "A", B: "B", T: "T", P: "P" },
    },
  );
}

/** 09 접선과 현 ⑵ — 접점 B 의 접선과 원에 내접하는 □ABCD, 대각선 AC. */
function figTangentCyclic(o: {
  d: number;
  t: number;
  angles: Record<string, unknown>;
}): FigureSpec {
  const R = 74;
  const O: Pt = [120, 96];
  const tB = 270;
  const tC = tB - 2 * o.t;
  const tD = tB - 2 * o.t - (180 - o.d);
  const tA = tB - 2 * o.t - 2 * (180 - o.d);
  const B = polar(O, onAng(tB), R);
  return spec(
    {
      O,
      A: { type: "on_circle", circle: "k", angle: onAng(tA) },
      B: { type: "on_circle", circle: "k", angle: onAng(tB) },
      C: { type: "on_circle", circle: "k", angle: onAng(tC) },
      D: { type: "on_circle", circle: "k", angle: onAng(tD) },
      T: [B[0] - 70, B[1]] as Pt,
      U: [B[0] + 70, B[1]] as Pt,
    },
    { k: { center: "O", radius: R } },
    {
      segments: {
        TU: ["T", "U"],
        AB: ["A", "B"],
        BC: ["B", "C"],
        CD: ["C", "D"],
        DA: ["D", "A"],
        AC: ["A", "C"],
      },
      angles: o.angles,
      labels: { A: "A", B: "B", C: "C", D: "D", T: "T", U: "T'" },
    },
  );
}

/** 10 할선이 원의 중심을 지날 때 — 지름 AB, 접점 T 의 접선이 직선 AB 와 P 에서 만난다. */
function figTangentDiameter(o: {
  a: number;
  angles: Record<string, unknown>;
}): FigureSpec {
  const R = 58;
  const O: Pt = [140, 96];
  const tA = onAng(180);
  const tB = onAng(0);
  const tT = onAng(180 - 2 * o.a);
  const px = O[0] - R / Math.cos(2 * o.a * DEG);
  const P: Pt = [px, O[1]];
  return spec(
    {
      O,
      A: { type: "on_circle", circle: "k", angle: tA },
      B: { type: "on_circle", circle: "k", angle: tB },
      T: { type: "on_circle", circle: "k", angle: tT },
      P,
    },
    { k: { center: "O", radius: R } },
    {
      segments: {
        PB: ["P", "B"],
        AT: ["A", "T"],
        BT: ["B", "T"],
        PT: ["P", "T"],
      },
      angles: o.angles,
      labels: { A: "A", B: "B", T: "T", P: "P", O: "O" },
    },
  );
}

/**
 * 11 한 점에서 원에 그은 두 접선 — 원 O 는 △ABC 의 내접원이면서 △DEF 의 외접원이다.
 * 접점 D(변 AB) · E(변 BC) · F(변 CA). 꼭짓점은 두 접선의 교점으로 **계산**한다.
 */
function figIncircleTriangle(o: {
  arcDE: number;
  arcEF: number;
  arcFD: number;
  angles: Record<string, unknown>;
}): FigureSpec {
  const R = 46;
  const O: Pt = [110, 104];
  const tD = 150;
  const tE = tD - o.arcDE;
  const tF = tE - o.arcEF;
  // 접선 두 개는 두 접점 사이 호의 중점 방향으로 R/cos(호/2) 만큼 떨어진 곳에서 만난다.
  const vertex = (t1: number, arc: number): Pt =>
    polar(O, onAng(t1 - arc / 2), R / Math.cos((arc / 2) * DEG));
  const B = vertex(tD, o.arcDE);
  const C = vertex(tE, o.arcEF);
  const A = vertex(tF, o.arcFD);
  return spec(
    {
      O,
      D: { type: "on_circle", circle: "k", angle: onAng(tD) },
      E: { type: "on_circle", circle: "k", angle: onAng(tE) },
      F: { type: "on_circle", circle: "k", angle: onAng(tF) },
      A,
      B,
      C,
    },
    { k: { center: "O", radius: R } },
    {
      segments: {
        AB: ["A", "B"],
        BC: ["B", "C"],
        CA: ["C", "A"],
        DE: ["D", "E"],
        EF: ["E", "F"],
        FD: ["F", "D"],
      },
      angles: o.angles,
      labels: { A: "A", B: "B", C: "C", D: "D", E: "E", F: "F", O: "O" },
    },
  );
}

/**
 * 12 두 원에서 접선과 현 ⑴ — 두 원이 점 T 에서 **밖으로** 접하고 공통접선 EF 가 세로다.
 * 세 점 A·T·D 와 세 점 B·T·C 가 각각 한 직선 위에 있다.
 */
function figTwoCirclesTangent(o: {
  a: number;
  c: number;
  angles: Record<string, unknown>;
}): FigureSpec {
  const r1 = 52;
  const r2 = 46;
  const T: Pt = [120, 100];
  const O1: Pt = [T[0] - r1, T[1]];
  const O2: Pt = [T[0] + r2, T[1]];
  const uB = [Math.cos((-90 - o.a) * DEG), -Math.sin((-90 - o.a) * DEG)] as Pt;
  const uD = [Math.cos((o.c - 90) * DEG), -Math.sin((o.c - 90) * DEG)] as Pt;
  const B = chordOther(O1, T, uB);
  const D = chordOther(O2, T, uD);
  const A = chordOther(O1, T, mul(uD, -1));
  const C = chordOther(O2, T, mul(uB, -1));
  const E: Pt = [T[0], T[1] - 76];
  const F: Pt = [T[0], T[1] + 76];
  return spec(
    { O: O1, W: O2, T, A, B, C, D, E, F },
    { k: { center: "O", radius: r1 }, k2: { center: "W", radius: r2 } },
    {
      segments: {
        EF: ["E", "F"],
        AD: ["A", "D"],
        BC: ["B", "C"],
        AB: ["A", "B"],
        CD: ["C", "D"],
      },
      angles: o.angles,
      labels: {
        A: "A",
        B: "B",
        C: "C",
        D: "D",
        T: "T",
        E: "E",
        F: "F",
      },
    },
  );
}

/**
 * 13 두 원에서 접선과 현 ⑵ — 작은 원이 큰 원에 **안으로** 접하고 공통접선 PQ 가 세로다.
 * 세 점 A·C·T 와 세 점 B·D·T 가 각각 한 직선 위에 있다.
 */
function figTwoCirclesInner(o: {
  a: number;
  b: number;
  angles: Record<string, unknown>;
}): FigureSpec {
  const r1 = 68;
  const r2 = 36;
  const T: Pt = [58, 100];
  const O1: Pt = [T[0] + r1, T[1]];
  const O2: Pt = [T[0] + r2, T[1]];
  const uA = [Math.cos((90 - o.b) * DEG), -Math.sin((90 - o.b) * DEG)] as Pt;
  const uB = [Math.cos((o.a - 90) * DEG), -Math.sin((o.a - 90) * DEG)] as Pt;
  const A = chordOther(O1, T, uA);
  const B = chordOther(O1, T, uB);
  const C = chordOther(O2, T, uA);
  const D = chordOther(O2, T, uB);
  const P: Pt = [T[0], T[1] - 74];
  const Q: Pt = [T[0], T[1] + 74];
  return spec(
    { O: O1, W: O2, T, A, B, C, D, P, Q },
    { k: { center: "O", radius: r1 }, k2: { center: "W", radius: r2 } },
    {
      segments: {
        PQ: ["P", "Q"],
        TA: ["T", "A"],
        TB: ["T", "B"],
        AB: ["A", "B"],
        CD: ["C", "D"],
      },
      angles: o.angles,
      labels: {
        A: "A",
        B: "B",
        C: "C",
        D: "D",
        T: "T",
        P: "P",
        Q: "Q",
      },
    },
  );
}

/**
 * 14 삼각형의 닮음 — 지름 AB 인 원의 접선(접점 C)에 A(또는 B)에서 내린 수선의 발 P.
 * `fromA=false` 면 B 에서 내린다.
 */
function figTangentSimilar(o: {
  cosPhi: number;
  fromA: boolean;
  angles: Record<string, unknown>;
}): FigureSpec {
  const R = 58;
  const O: Pt = [116, 104];
  const tA = onAng(180);
  const tB = onAng(0);
  const phi = Math.acos(Math.max(-1, Math.min(1, o.cosPhi))) / DEG;
  const tC = onAng(o.fromA ? 180 - phi : phi);
  const A = polar(O, tA, R);
  const B = polar(O, tB, R);
  const C = polar(O, tC, R);
  const tanDir = unit([-(C[1] - O[1]), C[0] - O[0]]);
  const src = o.fromA ? A : B;
  const P = foot(src, C, add(C, tanDir));
  return spec(
    {
      O,
      A: { type: "on_circle", circle: "k", angle: tA },
      B: { type: "on_circle", circle: "k", angle: tB },
      C: { type: "on_circle", circle: "k", angle: tC },
      P,
    },
    { k: { center: "O", radius: R } },
    {
      segments: {
        AB: ["A", "B"],
        AC: ["A", "C"],
        BC: ["B", "C"],
        CP: ["C", "P"],
        SP: [o.fromA ? "A" : "B", "P"],
      },
      angles: o.angles,
      labels: { A: "A", B: "B", C: "C", P: "P", O: "O" },
    },
  );
}

type Body = Pick<
  MidProblem,
  "content" | "answer" | "solution" | "facts" | "figureSpec"
>;

/**
 * 유형별 «치수를 다나» 정책 (원장님 확정 2026-08-25 — `m3Tangent.ts` 와 같은 규약).
 * 없음 = 안 단다 · 배열 = 단다, 그 이름들만 뺀다.
 *
 * `14` 의 `AB` 는 **원의 지름**이다 — 「지름은 치수선이 너저분해지니까 빼고」.
 */
const DIM_POLICY: Record<string, readonly string[] | undefined> = {
  "14": ["AB"],
};

function def(
  rpmType: string,
  rpmTitle: string,
  difficulty: MidProblem["difficulty"],
  gen: (rng: Rng) => Body,
): TypeEntry {
  return {
    key: `${BOOK}|${MINOR}|${rpmType}`,
    grade: GRADE,
    unitSection: SEC,
    rpmBook: BOOK,
    rpmMinor: MINOR,
    rpmType,
    rpmTitle,
    handler: (rng) => {
      const body = gen(rng);
      return {
        grade: GRADE,
        unitSection: SEC,
        rpmBook: BOOK,
        rpmMinor: MINOR,
        rpmType,
        rpmTitle,
        difficulty,
        ...body,
        figureSpec:
          DIM_POLICY[rpmType] === undefined
            ? body.figureSpec
            : attachGivenDims(
                body.figureSpec,
                body.content,
                body.answer,
                DIM_POLICY[rpmType],
              ),
      };
    },
  };
}
