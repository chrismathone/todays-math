/**
 * 중3 「이차함수」 — RPM 3-1 «이차함수의 그래프 ⑴» 01~21 + «이차함수의 그래프 ⑵» 01~21.
 *
 * 물결 5A 세션 소유 모듈 (docs/planning/tracks/middle-engine-wave3-plan.md §12).
 * 합본 `m3.ts` 에는 두 줄(import + spread)만 더한다.
 *
 * ## 생성 규칙 — **답을 먼저 정하고 발문을 만든다**
 *
 * 꼭짓점 $(p, q)$ 와 $x^2$ 의 계수 $a$ 를 **먼저** 고르고, 지나는 점·일반형 계수를
 * 그것에서 역산한다. 그래서 시험은 **표준형 ↔ 일반형** 두 꼴을 각각 파싱해
 * 여러 정수 점에 대입한다 — 두 꼴이 서로의 검산이다(5A 브리프).
 *
 * ## 부호를 조심한 자리 (브리프가 미리 짚은 것)
 *
 * - `y=a(x-p)^2+q` 에서 `p>0` 이면 그래프는 **오른쪽**으로 간다. 발문의
 *   「$x$축의 방향으로 $-3$만큼」과 식의 `(x+3)` 이 맞는지 `facts` 에 이동량을
 *   따로 실어 시험이 **원함수를 옮겨서** 다시 확인한다(⑴09·⑴12·⑴19·⑵07).
 * - **최댓값·최솟값은 `a` 의 부호가 가른다.** `a>0` 이면 최솟값만, `a<0` 이면
 *   최댓값만 있다. `facts.isMax` 를 싣고 시험이 `(a<0) === isMax` 를 잠근다
 *   (⑵14·⑵15·⑵16·⑵20).
 * - **평행이동과 대칭이동을 한 유형에서 섞지 않는다.** 대칭이동은 ⑴05 에만 있고,
 *   그 유형에는 평행이동이 없다.
 * - 근이 무리수가 되는 갈래는 내지 않는다 — $x$축 교점을 묻는 유형(⑵05·⑵08·⑵13)은
 *   **정수 근에서 역산**해 만든다(`\sqrt{8}` 이 지면에 나갈 자리가 없다).
 *

/** 조건을 만족할 때까지 다시 뽑는다 — 못 찾으면 던진다(조용히 기본값 금지). */
function attempt<T>(tries: number, f: () => T | null): T {
  for (let i = 0; i < tries; i += 1) {
    const v = f();
    if (v !== null) return v;
  }
  throw new Error("유형 생성 재시도 초과 — 매개변수 범위를 확인하십시오");
}

/* ────────────────────────────────────────────────────────────────────────
 * 유리수 — 이차항의 계수 `a` 가 분수인 갈래가 있다(⑴ 전반)
 * ──────────────────────────────────────────────────────────────────────── */

/** `[분자, 분모]` — 분모는 양수, 기약. */
export type Rat = readonly [number, number];

function gcd(a: number, b: number): number {
  return b === 0 ? Math.abs(a) : gcd(b, Math.abs(a) % Math.abs(b));
}

function rat(n: number, d: number): Rat {
  if (d === 0) throw new Error("분모가 0 입니다");
  let nn = n;
  let dd = d;
  if (dd < 0) {
    nn = -nn;
    dd = -dd;
  }
  const g = gcd(nn, dd) || 1;
  return [nn / g, dd / g];
}

const rInt = (n: number): Rat => [n, 1];
const rVal = (r: Rat): number => r[0] / r[1];
const rNeg = (r: Rat): Rat => [-r[0], r[1]];
const rAdd = (a: Rat, b: Rat): Rat =>
  rat(a[0] * b[1] + b[0] * a[1], a[1] * b[1]);
const rMul = (a: Rat, b: Rat): Rat => rat(a[0] * b[0], a[1] * b[1]);

/** 기약분수 LaTeX. 정수면 정수, 음수면 앞에 `-`. 대분수는 쓰지 않는다(D-73). */
function ratTex(r: Rat): string {
  if (r[1] === 1) return String(r[0]);
  return r[0] < 0 ? `-\\dfrac{${-r[0]}}{${r[1]}}` : `\\dfrac{${r[0]}}{${r[1]}}`;
}

/**
 * 항 하나 — 계수 `r`, 문자 부분 `v`. `lead` 면 맨 앞이라 `+` 를 생략한다.
 * 계수가 $0$ 이면 빈 문자열, $\pm1$ 이면 수를 안 적는다(`1x^2` 가 지면에 못 나간다).
 */
function termTex(r: Rat, v: string, lead = false): string {
  if (r[0] === 0) return "";
  const sign = r[0] < 0 ? "-" : lead ? "" : "+";
  const abs: Rat = [Math.abs(r[0]), r[1]];
  if (v === "") return `${sign}${ratTex(abs)}`;
  if (abs[0] === 1 && abs[1] === 1) return `${sign}${v}`;
  return `${sign}${ratTex(abs)}${v}`;
}

/** `(x-p)^2` — `p` 가 $0$ 이면 `x^2`. 부호를 여기 한 곳에서만 뒤집는다. */
function sqTex(p: number): string {
  if (p === 0) return "x^2";
  return `(x${p > 0 ? "-" : "+"}${Math.abs(p)})^2`;
}

/** `(x-r)` — 근이 $0$ 이면 괄호 없이 `x`. `(x-0)` 이 지면에 나가면 안 된다. */
function rootTex(r: number): string {
  return r === 0 ? "x" : `(x${r > 0 ? "-" : "+"}${Math.abs(r)})`;
}

/** 인수분해 꼴 `a(x-r_1)(x-r_2)` — 계수는 `termTex` 가 맡아 `1`·`-1` 을 안 적는다. */
function factoredTex(a: Rat, r1: number, r2: number): string {
  return termTex(a, `${rootTex(r1)}${rootTex(r2)}`, true);
}

/** 미지수 `a` 를 그대로 둔 인수분해 꼴 — 「$y=a(x-r_1)(x-r_2)$ 로 놓는다」. */
function factoredWithA(r1: number, r2: number): string {
  return `a${rootTex(r1)}${rootTex(r2)}`;
}

/** 표준형 우변 `a(x-p)^2+q`. */
function stdRhs(a: Rat, p: number, q: number): string {
  return `${termTex(a, sqTex(p), true)}${termTex(rInt(q), "")}`;
}

/**
 * `표준형=일반형` — 꼭짓점이 $y$축 위($p=0$)면 두 꼴이 같아 `x^2-2=x^2-2` 가 되므로 한 번만 적는다.
 */
function stdEqGen(std: string, gen: string): string {
  return std === gen ? gen : `${std}=${gen}`;
}

/** 일반형 우변 `ax^2+bx+c`. */
function genRhs(A: Rat, B: Rat, C: Rat): string {
  const head = termTex(A, "x^2", true);
  return `${head}${termTex(B, "x")}${termTex(C, "")}`;
}

/** 표준형 → 일반형 계수. 시험은 두 꼴을 각각 파싱해 값으로 견준다. */
function expand(a: Rat, p: number, q: number): { A: Rat; B: Rat; C: Rat } {
  return {
    A: a,
    B: rMul(a, rInt(-2 * p)),
    C: rAdd(rMul(a, rInt(p * p)), rInt(q)),
  };
}

/**
 * 식 안에서 다른 수 뒤에 오는 수 — **음수만** 괄호로 싼다.
 * `2	imes(2)^2` 처럼 양수를 싸면 지면이 흉하다(값 검사는 통과한다 — 2026-08-24).
 */
function baseTex(v: number): string {
  return v < 0 ? `(${v})` : String(v);
}

/** 좌표 `(m, n)`. */
function pt(m: number, n: number): string {
  return `(${m}, ${n})`;
}

/** `이동량 만큼` 앞에 붙는 수 — 음수도 그대로 적는다(발문의 「$-3$만큼」). */
function shiftTex(v: number): string {
  return String(v);
}

/**
 * 수 뒤의 조사 — 끝 자리를 한자 수사로 읽었을 때 받침이 있는가로 가른다(`m2Josa` 와 같은 규칙).
 * 「(으)로」는 여기 두지 않고 `m2Josa.euroRo` 를 쓴다 — 받침 유무만 보면 ㄹ 받침(1·7·8)에
 * 「$7$으로」가 나간다(일로·칠로·팔로가 맞다).
 */
function vowelFinal(n: number): boolean {
  const d = Math.abs(n) % 10;
  return d === 2 || d === 4 || d === 5 || d === 9;
}

/** 「$3$과」·「$4$와」 */
function waGwaTex(n: number): "와" | "과" {
  return vowelFinal(n) ? "와" : "과";
}

/**
 * 평행이동 설명의 「$x$ 대신 $x-3$」 — 부호를 여기 한 곳에서만 뒤집는다.
 * `x-(-3)` 이 지면에 나가면 안 된다.
 */
function shiftedVarTex(v: string, t: number): string {
  return t >= 0 ? `${v}-${t}` : `${v}+${-t}`;
}

/** 엔진이 요구하는 것보다 **여유를 두고** 잡는다 — 경계에 딱 붙으면 흠집으로 보인다. */
const FIG_MIN_ARC = 1.6; // 엔진 하한 1.2
const FIG_MIN_RISE = 1.4; // 엔진 하한 1.0
/**
 * 꼭짓점 **양쪽**으로 보여야 하는 칸수.
 *
 * 엔진은 「꼭짓점이 범위 안」까지만 본다. 그런데 꼭짓점이 범위의 **모서리**에 있으면
 * 팔이 한쪽만 그려져 **포물선으로 안 읽힌다** — 「그림에서 $p$, $q$ 의 부호를
 * 읽으시오」류는 그 순간 답을 못 구하는 문항이 된다(2026-08-24 실렌더 육안:
 * ⑴20·⑵09 가 그렇게 났다). 그래서 여기서 한 겹 더 좁힌다.
 */
const FIG_MIN_SIDE = 1.0;

/** 범위 후보 — 원점을 품고 폭 4~12칸, 필요한 값이 다 들어가는 것. */
function rangeCandidates(needLo: number, needHi: number): [number, number][] {
  const out: [number, number][] = [];
  for (let lo = -12; lo <= -1; lo += 1) {
    for (let hi = 1; hi <= 12; hi += 1) {
      const span = hi - lo;
      if (span < 4 || span > 12) continue;
      if (lo > needLo || hi < needHi) continue;
      out.push([lo, hi]);
    }
  }
  // 폭이 좁을수록·원점이 가운데일수록 먼저 — 그림이 프레임을 채운다.
  out.sort(
    (u, v) =>
      u[1] - u[0] - (v[1] - v[0]) ||
      Math.abs(u[0] + u[1]) - Math.abs(v[0] + v[1]),
  );
  return out;
}

/**
 * 표식 숫자가 **원점 글자 `O` 와 겹치는 자리**인가.
 *
 * 엔진은 표식의 $x$ 숫자를 눈금에서 $5$ 만큼 **원점 쪽으로** 밀어 적고
 * (`_cp_mark_parts`), `O` 는 축 왼쪽 아래 $7$ 에 적는다. 그래서 $x=\pm1$ 인 표식은
 * 칸 크기가 작을 때 `O` 위에 올라앉아 지면에서 **`-10` 으로 읽힌다**
 * (2026-08-24 실렌더 육안 — ⑵08·⑴18 에서 실제로 났다). $y=-1$ 도 같은 자리다.
 *
 * 엔진은 5E 소유라 여기서 고치지 않는다 — 그런 자리를 **안 쓴다**.
 * (겹침 회피는 엔진에 있어야 할 규칙이므로 main 세션에 알린다.)
 */
function markCollidesWithOrigin(mx: number, my: number): boolean {
  return Math.abs(mx) === 1 || my === -1;
}

/**
 * 포물선 그림 한 장. 못 그리는 조합이면 **`null`** 을 돌려 갈래를 다시 뽑게 한다
 * (조용히 그림 없이 내보내지 않는다).
 *
 * 여기서 재는 것은 엔진이 던지는 조건과 **같은 성질**이되 여유가 더 크다 —
 * 정본은 엔진이고, 시험이 이 스펙을 **실제 엔진으로 그려서** 확인한다.
 *
 * `coverX` 를 주면 **그 구간이 보이는 토막 안에** 들어와야 한다. 「$y$축과의 교점으로
 * $c$ 의 부호를 읽으시오」류는 곡선이 $x=0$ 에서 상자 밖으로 나가 있으면 **발문이
 * 묻는 것이 그림에 없다** — 값 검사로는 안 잡히고 실렌더를 눈으로 봐야 나온다
 * (2026-08-24 ⑵09 에서 실제로 났다).
 */
function parabolaFig(
  a: Rat,
  p: number,
  q: number,
  marks: readonly (readonly [number, number])[],
  coverX?: readonly [number, number],
): Record<string, unknown> | null {
  const av = rVal(a);
  if (Math.abs(av) < 1 / 8 || Math.abs(av) > 8) return null;
  if (Math.abs(p) > 12 || Math.abs(q) > 20) return null;
  for (const [mx, my] of marks) {
    if (!Number.isInteger(mx) || !Number.isInteger(my)) return null;
    if (mx === 0 && my === 0) return null; // 원점은 O 로 이미 적힌다
    if (markCollidesWithOrigin(mx, my)) return null;
    if (Math.abs(my - (av * (mx - p) * (mx - p) + q)) > 1e-9) return null;
  }
  const f = (x: number): number => av * (x - p) * (x - p) + q;
  const xsNeed = [p, ...marks.map((m) => m[0])];
  const ysNeed = [q, ...marks.map((m) => m[1])];
  if (coverX) {
    xsNeed.push(coverX[0], coverX[1]);
    ysNeed.push(f(coverX[0]), f(coverX[1]));
    if (!ysNeed.every(Number.isFinite)) return null;
  }
  const yCands = rangeCandidates(
    Math.floor(Math.min(...ysNeed)),
    Math.ceil(Math.max(...ysNeed)),
  );
  const xCands = rangeCandidates(Math.min(...xsNeed), Math.max(...xsNeed));
  for (const [ylo, yhi] of yCands) {
    // 꼭짓점 위(아래)로 남은 세로 여유가 곧 보이는 팔의 길이를 정한다.
    const head = av > 0 ? yhi - q : q - ylo;
    if (head <= 0) continue;
    const w = Math.sqrt(head / Math.abs(av));
    for (const [xlo, xhi] of xCands) {
      const ra = Math.max(xlo, p - w);
      const rb = Math.min(xhi, p + w);
      if (rb - ra < FIG_MIN_ARC) continue;
      // 꼭짓점이 모서리에 붙으면 팔이 한쪽만 그려져 포물선으로 안 읽힌다.
      if (p - ra < FIG_MIN_SIDE || rb - p < FIG_MIN_SIDE) continue;
      const rise = Math.max(Math.abs(f(ra) - q), Math.abs(f(rb) - q));
      if (rise < FIG_MIN_RISE) continue;
      if (coverX && (ra > coverX[0] + 1e-9 || rb < coverX[1] - 1e-9)) continue;
      return {
        version: "elem-1",
        kind: "curveGraph",
        curves: [{ type: "quadratic", a: av, p, q }],
        xRange: [xlo, xhi],
        yRange: [ylo, yhi],
        ...(marks.length > 0 ? { marks: marks.map((m) => [m[0], m[1]]) } : {}),
      };
    }
  }
  return null;
}

/** 표식은 `choiceCountRule` 한 곳에서 온다 — 세 모듈이 각자 적으면 한쪽만 고쳐진다. */
const MARKS = CHOICE_MARKS;

function listTex(items: readonly string[]): string {
  return items.map((t, i) => `${MARKS[i]!} ${t}`).join("\n");
}

function marksOf(flags: readonly boolean[]): string {
  const out: string[] = [];
  flags.forEach((v, i) => {
    if (v) out.push(MARKS[i]!);
  });
  return out.join(", ");
}

/** 배열에서 `count` 개를 섞어 뽑는다 — Fisher–Yates(앞쪽 편중 방지). */
function pickSome<T>(rng: Rng, items: readonly T[], count: number): T[] {
  if (count > items.length) throw new Error("뽑을 개수가 전체보다 많습니다");
  const bag = [...items];
  for (let i = bag.length - 1; i > 0; i -= 1) {
    const j = intBetween(rng, 0, i);
    const t = bag[i]!;
    bag[i] = bag[j]!;
    bag[j] = t;
  }
  return bag.slice(0, count);
}

/** 서로 다른 색인 `count` 개 — Fisher–Yates 로 섞는다(앞쪽 편중 방지). */
function distinctIndices(rng: Rng, total: number, count: number): number[] {
  return pickSome(
    rng,
    Array.from({ length: total }, (_, i) => i),
    count,
  );
}

/**
 * 「모두 고르시오」 보기 다섯 — **참 2개 + 거짓 3개**를 골라 담아 섞는다
 * (원장님 2026-08-25, `CHOOSE_ALL_ANSWER_COUNT`).
 *
 * 🔴 **거르지 않고 만든다.** 예전에는 아홉 중 다섯을 그냥 뽑고 참이 1~4개면
 *    받았다. 그런데 이 풀들은 참이 6~7개라, 계수 부호에 따라 **거짓이 두 개뿐인
 *    구간**이 생긴다 — 거기서는 「참 2개」가 **아예 나올 수 없다**(실측
 *    2026-08-25: ⑴06 은 2개가 7.0%, ⑵06 은 14.0%). 그래서 거르는 것으로는
 *    못 고치고, 풀에 **거짓 명제를 더해** 언제나 셋을 고를 수 있게 했다.
 *    풀이 모자라면 조용히 보기 넷을 내지 않고 **던진다.**
 */
function chooseFivePropIndices(
  rng: Rng,
  props: readonly { ok: boolean }[],
): number[] {
  const trueIdx = props.map((_p, i) => i).filter((i) => props[i]!.ok);
  const falseIdx = props.map((_p, i) => i).filter((i) => !props[i]!.ok);
  const need = CHOOSE_ALL_ANSWER_COUNT;
  if (trueIdx.length < need || falseIdx.length < 5 - need) {
    throw new Error(
      `보기 풀이 모자랍니다: 참 ${trueIdx.length}(필요 ${need}) · 거짓 ${falseIdx.length}(필요 ${5 - need})`,
    );
  }
  return pickSome(
    rng,
    [...pickSome(rng, trueIdx, need), ...pickSome(rng, falseIdx, 5 - need)],
    5,
  );
}

/** `a(x^2-2px)+C` — 계수로 묶은 꼴. */
function groupTex(a: number, p: number, C: number): string {
  const body = `(x^2${termTex(rInt(-2 * p), "x")})`;
  return `${termTex(rInt(a), body, true)}${termTex(rInt(C), "")}`;
}

/** 완전제곱식으로 고친 중간 꼴. `|a|=1` 이면 중괄호를 안 쓴다. */
function squareTex(a: number, p: number, C: number): string {
  if (Math.abs(a) === 1) {
    return `${termTex(rInt(a), sqTex(p), true)}${termTex(rInt(-a * p * p), "")}${termTex(rInt(C), "")}`;
  }
  const body = `\\{${sqTex(p)}-${p * p}\\}`;
  return `${termTex(rInt(a), body, true)}${termTex(rInt(C), "")}`;
}

/**
 * 일반형 → 표준형 풀이 줄. `a`·`p` 는 정수다(그래야 묶는 계수 $-2p$ 가 정수라
 * 지면에 분수가 안 낀다) — ⑵ 계열은 전부 정수 `a` 로 만든다.
 */
function toStdLines(a: number, p: number, q: number): string[] {
  if (p === 0) {
    throw new Error("일반형 → 표준형 풀이는 일차항이 있는 갈래에만 씁니다");
  }
  const C = a * p * p + q;
  const std = stdRhs(rInt(a), p, q);
  // ⚠️ **항상 두 줄이다.** $|a|=1$ 일 때 묶는 줄을 건너뛰면 해설이 한 줄로 뭉쳐
  //    「단계 사이 줄바꿈」 가드에 걸린다(2026-08-24 시험이 잡았다).
  if (Math.abs(a) === 1) {
    return [
      `$x$의 계수의 절반의 제곱인 $${p * p}$${eulReul(p * p)} 더하고 빼면 $y=${squareTex(a, p, C)}$이다.`,
      `따라서 $y=${std}$이다.`,
    ];
  }
  return [
    `$x^2$의 계수 $${a}$${euroRo(a)} 묶으면 $y=${groupTex(a, p, C)}$이다.`,
    `완전제곱식으로 고치면 $y=${squareTex(a, p, C)}$이므로 $y=${std}$이다.`,
  ];
}
